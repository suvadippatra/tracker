fun EditHotelModal(
    hotel: com.example.data.model.HotelEntity,
    viewModel: FinancialViewModel,
    onDismiss: () -> Unit
) {
    var editHotelName by remember { mutableStateOf(hotel.name) }
    var editHotelAdvanced by remember { mutableStateOf(hotel.advancedPayment.toString()) }
    var editPhone by remember { mutableStateOf(hotel.phoneNumber) }
    var editLocation by remember { mutableStateOf(hotel.location) }
    var editUpiId by remember { mutableStateOf(hotel.upiId) }
    var editQrCodeUri by remember { mutableStateOf(hotel.qrCodeUri) }
    var showAddMenu by remember { mutableStateOf(false) }

    ModalBottomSheet(
        onDismissRequest = onDismiss,
        containerColor = getDialogBackground(),
        sheetState = rememberModalBottomSheetState(skipPartiallyExpanded = true)
    ) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .verticalScroll(rememberScrollState())
                .padding(horizontal = 24.dp, vertical = 16.dp)
        ) {
            Text("Edit Hotel Info", color = AppTextColor(), fontSize = 20.sp, fontWeight = FontWeight.Bold)
            Spacer(modifier = Modifier.height(16.dp))

            OutlinedTextField(
                value = editHotelName,
                onValueChange = { editHotelName = it },
                label = { Text("Hotel Name", color = AppTextColor().copy(alpha = 0.6f)) },
                colors = OutlinedTextFieldDefaults.colors(
                    focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor(),
                    focusedBorderColor = AccentCyan, unfocusedBorderColor = AppTextColor().copy(alpha = 0.3f)
                ),
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(modifier = Modifier.height(12.dp))
            OutlinedTextField(
                value = editHotelAdvanced,
                onValueChange = { editHotelAdvanced = it },
                label = { Text("Advanced Payment (₹)", color = AppTextColor().copy(alpha = 0.6f)) },
                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                colors = OutlinedTextFieldDefaults.colors(
                    focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor(),
                    focusedBorderColor = AccentCyan, unfocusedBorderColor = AppTextColor().copy(alpha = 0.3f)
                ),
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(modifier = Modifier.height(12.dp))
            OutlinedTextField(
                value = editPhone,
                onValueChange = { editPhone = it },
                label = { Text("Mobile Number", color = AppTextColor().copy(alpha = 0.6f)) },
                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Phone),
                colors = OutlinedTextFieldDefaults.colors(
                    focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor(),
                    focusedBorderColor = AccentCyan, unfocusedBorderColor = AppTextColor().copy(alpha = 0.3f)
                ),
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(modifier = Modifier.height(12.dp))
            OutlinedTextField(
                value = editLocation,
                onValueChange = { editLocation = it },
                label = { Text("Location", color = AppTextColor().copy(alpha = 0.6f)) },
                colors = OutlinedTextFieldDefaults.colors(
                    focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor(),
                    focusedBorderColor = AccentCyan, unfocusedBorderColor = AppTextColor().copy(alpha = 0.3f)
                ),
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(modifier = Modifier.height(12.dp))
            OutlinedTextField(
                value = editUpiId,
                onValueChange = { editUpiId = it },
                label = { Text("UPI ID", color = AppTextColor().copy(alpha = 0.6f)) },
                colors = OutlinedTextFieldDefaults.colors(
                    focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor(),
                    focusedBorderColor = AccentCyan, unfocusedBorderColor = AppTextColor().copy(alpha = 0.3f)
                ),
                modifier = Modifier.fillMaxWidth()
            )
            Spacer(modifier = Modifier.height(12.dp))
            OutlinedTextField(
                value = editQrCodeUri,
                onValueChange = { editQrCodeUri = it },
                label = { Text("QR Code/UPI Scan URI", color = AppTextColor().copy(alpha = 0.6f)) },
                colors = OutlinedTextFieldDefaults.colors(
                    focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor(),
                    focusedBorderColor = AccentCyan, unfocusedBorderColor = AppTextColor().copy(alpha = 0.3f)
                ),
                modifier = Modifier.fillMaxWidth()
            )
            
            Spacer(modifier = Modifier.height(24.dp))
            
            Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.SpaceBetween, verticalAlignment = Alignment.CenterVertically) {
                Text(text = "Menu Chart", color = AppTextColor(), fontSize = 18.sp, fontWeight = FontWeight.Bold)
                Icon(
                    imageVector = Icons.Default.Add,
                    contentDescription = "Add Menu",
                    tint = AccentCyan,
                    modifier = Modifier.size(24.dp).clickable { showAddMenu = !showAddMenu }
                )
            }
            Text("Edit menu chart and items. You can add Day, Night, or Common meals.", color = AppTextColor().copy(alpha = 0.6f), fontSize = 12.sp)
            Spacer(modifier = Modifier.height(16.dp))

            // Menu Items List
            val menuItems by viewModel.allMenuItems.collectAsState(initial = emptyList())
            
            Column(modifier = Modifier.fillMaxWidth().heightIn(max = 300.dp).verticalScroll(rememberScrollState())) {
                menuItems.forEach { menuItem ->
                    Row(
                        modifier = Modifier.fillMaxWidth().padding(vertical = 8.dp).background(AppTextColor().copy(alpha = 0.05f), RoundedCornerShape(8.dp)).padding(8.dp),
                        horizontalArrangement = Arrangement.SpaceBetween,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Column {
                            Text("${menuItem.emoji} ${menuItem.englishName}", color = AppTextColor(), fontWeight = FontWeight.Bold)
                            Text("₹${menuItem.price} • ${menuItem.type}", color = AppTextColor().copy(alpha = 0.7f), fontSize = 12.sp)
                        }
                        Row {
                            Icon(Icons.Default.Edit, contentDescription = "Edit", tint = AccentCyan, modifier = Modifier.size(20.dp).clickable { /* TODO: inline edit */ })
                            Spacer(modifier = Modifier.width(16.dp))
                            Icon(Icons.Default.Delete, contentDescription = "Delete", tint = AccentPink, modifier = Modifier.size(20.dp).clickable { viewModel.deleteMenu(menuItem) })
                        }
                    }
                }
            }
            
            if (showAddMenu) {
                var newName by remember { mutableStateOf("") }
                var newPrice by remember { mutableStateOf("") }
                var newEmoji by remember { mutableStateOf("🍽️") }
                var newType by remember { mutableStateOf("Day") } // Day, Night, Common
                
                GlassCard(modifier = Modifier.fillMaxWidth().padding(vertical = 8.dp)) {
                    Column(modifier = Modifier.padding(16.dp)) {
                        Text("Add New Menu Item", color = AccentCyan, fontWeight = FontWeight.Bold)
                        Spacer(modifier = Modifier.height(8.dp))
                        Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                            OutlinedTextField(
                                value = newEmoji, onValueChange = { newEmoji = it },
                                label = { Text("Emoji/URI", color = AppTextColor().copy(alpha = 0.5f)) },
                                modifier = Modifier.weight(0.3f),
                                colors = OutlinedTextFieldDefaults.colors(focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor())
                            )
                            OutlinedTextField(
                                value = newName, onValueChange = { newName = it },
                                label = { Text("Name", color = AppTextColor().copy(alpha = 0.5f)) },
                                modifier = Modifier.weight(0.7f),
                                colors = OutlinedTextFieldDefaults.colors(focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor())
                            )
                        }
                        Spacer(modifier = Modifier.height(8.dp))
                        Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.spacedBy(8.dp), verticalAlignment = Alignment.CenterVertically) {
                            OutlinedTextField(
                                value = newPrice, onValueChange = { newPrice = it },
                                label = { Text("Price (₹)", color = AppTextColor().copy(alpha = 0.5f)) },
                                keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Number),
                                modifier = Modifier.weight(0.5f),
                                colors = OutlinedTextFieldDefaults.colors(focusedTextColor = AppTextColor(), unfocusedTextColor = AppTextColor())
                            )
                            Row(modifier = Modifier.weight(0.5f).horizontalScroll(rememberScrollState()), horizontalArrangement = Arrangement.spacedBy(4.dp)) {
                                listOf("Day", "Night", "Common").forEach { typeOpt ->
                                    Box(
                                        modifier = Modifier
                                            .clip(RoundedCornerShape(8.dp))
                                            .background(if (newType == typeOpt) AccentCyan else Color.Transparent)
                                            .border(1.dp, AccentCyan, RoundedCornerShape(8.dp))
                                            .clickable { newType = typeOpt }
                                            .padding(horizontal = 8.dp, vertical = 4.dp)
                                    ) {
                                        Text(typeOpt, color = if (newType == typeOpt) Color.Black else AccentCyan, fontSize = 12.sp)
                                    }
                                }
                            }
                        }
                        Spacer(modifier = Modifier.height(16.dp))
                        Button(
                            onClick = {
                                val p = newPrice.toDoubleOrNull() ?: 0.0
                                if (newName.isNotBlank()) {
                                    viewModel.addMenu(com.example.data.model.MenuEntity(
                                        banglaName = newName,
                                        englishName = newName,
                                        price = p,
                                        isDayMenu = newType == "Day" || newType == "Common",
                                        isNightMenu = newType == "Night" || newType == "Common",
                                        isAddOn = false,
                                        emoji = newEmoji,
                                        type = newType
                                    ))
                                    showAddMenu = false
                                }
                            },
                            modifier = Modifier.fillMaxWidth(),
                            colors = ButtonDefaults.buttonColors(containerColor = AccentCyan)
                        ) {
                            Text("Save Menu Item", color = Color.Black)
                        }
                    }
                }
            }

            Spacer(modifier = Modifier.height(32.dp))
            Row(modifier = Modifier.fillMaxWidth(), horizontalArrangement = Arrangement.End) {
                TextButton(onClick = onDismiss) {
                    Text("Cancel", color = AppTextColor().copy(alpha = 0.6f))
                }
                Spacer(modifier = Modifier.width(8.dp))
                Button(
                    onClick = {
                        val adv = editHotelAdvanced.toDoubleOrNull() ?: 0.0
                        viewModel.editHotel(hotel.copy(
                            name = editHotelName,
                            advancedPayment = adv,
                            phoneNumber = editPhone,
                            location = editLocation,
                            upiId = editUpiId
                        ))
                        onDismiss()
                    },
                    colors = ButtonDefaults.buttonColors(containerColor = AccentCyan)
                ) {
                    Text("Save Changes", color = Color.Black)
                }
            }
            Spacer(modifier = Modifier.height(32.dp))
        }
    }
}

// ==========================================
// FULL-SCREEN GLASS ADD MEAL MODAL
// ==========================================
@Composable
fun AddMealFlowModal(
    viewModel: FinancialViewModel,
    onDismiss: () -> Unit
) {
    val date by viewModel.addMealDate.collectAsState()
    val isNight by viewModel.addMealIsNight.collectAsState()
    val quantities by viewModel.orderQuantities.collectAsState()
    val selectedMainId by viewModel.selectedMainDishId.collectAsState()
    val menuItems by viewModel.allMenuItems.collectAsState()
    val userProfile by viewModel.userProfile.collectAsState()

    val context = LocalContext.current
    val sdf = SimpleDateFormat("dd MMM yyyy, hh:mm a", Locale.getDefault())

    // Filter menu dynamically
    val filteredDishes = menuItems.filter {
        if (isNight) it.isNightMenu else it.isDayMenu
    }
    val mainDishes = filteredDishes.filter { !it.isAddOn }
    val addOnDishes = filteredDishes.filter { it.isAddOn }

    Dialog(
        onDismissRequest = onDismiss,
        properties = DialogProperties(
            usePlatformDefaultWidth = false,
            dismissOnBackPress = true,
            dismissOnClickOutside = false
        )
    ) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(getAppBackgroundGradient(userProfile.themeMode))
                .statusBarsPadding()
                .navigationBarsPadding()
        ) {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(16.dp)
            ) {
                // Modal Header
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    IconButton(
                        onClick = onDismiss,
                        modifier = Modifier.background(getGlassBorder(), CircleShape)
                    ) {
                        Icon(Icons.Default.ArrowBack, contentDescription = "Back", tint = AppTextColor())
                    }

                    Text(
                        text = "Add Meal Record",
                        color = AppTextColor(),
                        fontSize = 18.sp,
                        fontWeight = FontWeight.Bold
                    )

                    // Glowy Save Button
                    GlassButton(
                        onClick = {
                            viewModel.saveHotelTransaction()
                            onDismiss()
                        },
                        cornerRadius = 10.dp,
                        borderBrush = Brush.linearGradient(listOf(AccentCyan, AccentPink)),
                        modifier = Modifier.testTag("save_meal_button")
                    ) {
                        Text(
                            text = "Save",
                            color = AppTextColor(),
                            fontSize = 13.sp,
                            fontWeight = FontWeight.Bold
                        )
                    }
                }

                Spacer(modifier = Modifier.height(16.dp))

                Column(
                    modifier = Modifier
                        .weight(1f)
                        .verticalScroll(rememberScrollState())
                ) {
                    // TOP SECTION: DATE/TIME DRUM CONTROLS
                    GlassCard(modifier = Modifier.fillMaxWidth()) {
                        Row(
                            modifier = Modifier
                                .fillMaxWidth()
                                .padding(12.dp),
                            horizontalArrangement = Arrangement.SpaceBetween,
                            verticalAlignment = Alignment.CenterVertically
                        ) {
                            Column {
                                Text(
                                    text = "Date & Time Picked",
                                    color = AppTextColor().copy(alpha = 0.5f),
                                    fontSize = 11.sp
                                )
                                Text(
                                    text = sdf.format(Date(date)),
                                    color = AppTextColor(),
                                    fontSize = 15.sp,
                                    fontWeight = FontWeight.Bold
                                )
                            }

                            Row(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
                                // Date Picker Launcher
                                GlassButton(
                                    onClick = {
                                        val cal = Calendar.getInstance().apply { timeInMillis = date }
                                        DatePickerDialog(
                                            context,
                                            { _, y, m, d ->
                                                cal.set(Calendar.YEAR, y)
                                                cal.set(Calendar.MONTH, m)
                                                cal.set(Calendar.DAY_OF_MONTH, d)
                                                viewModel.setOrderDate(cal.timeInMillis)
                                            },
                                            cal.get(Calendar.YEAR),
                                            cal.get(Calendar.MONTH),
                                            cal.get(Calendar.DAY_OF_MONTH)
                                        ).show()
                                    },
                                    cornerRadius = 8.dp
                                ) {
                                    Icon(Icons.Default.CalendarToday, contentDescription = "Date", tint = AccentCyan, modifier = Modifier.size(16.dp))
                                }

                                // Time Picker Launcher
                                GlassButton(
                                    onClick = {
                                        val cal = Calendar.getInstance().apply { timeInMillis = date }
                                        TimePickerDialog(
                                            context,
                                            { _, hr, min ->
                                                cal.set(Calendar.HOUR_OF_DAY, hr)
                                                cal.set(Calendar.MINUTE, min)
                                                viewModel.setOrderDate(cal.timeInMillis)
                                            },
                                            cal.get(Calendar.HOUR_OF_DAY),
                                            cal.get(Calendar.MINUTE),
                                            false
                                        ).show()
                                    },
                                    cornerRadius = 8.dp
                                ) {
                                    Icon(Icons.Default.Schedule, contentDescription = "Time", tint = AccentCyan, modifier = Modifier.size(16.dp))
                                }
                            }
                        }
                    }

                    Spacer(modifier = Modifier.height(14.dp))

                    // DAY/NIGHT CUSTOM LIQUID SLIDING SWITCH
                    Row(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(vertical = 4.dp),
                        horizontalArrangement = Arrangement.SpaceBetween,
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Text(
                            text = "Menu Period Selection",
                            color = AppTextColor(),
                            fontSize = 15.sp,
                            fontWeight = FontWeight.Bold
                        )

                        // Highly styled liquid sliding track
                        val isAppLight = isAppLightMode()
                        Box(
                            modifier = Modifier
                                .width(130.dp)
                                .height(38.dp)
                                .clip(RoundedCornerShape(19.dp))
                                .background(if (isAppLight) Color(0x1A000000) else Color(0x1F000000))
                                .border(1.dp, if (isAppLight) Color(0xFFFFFFFF) else getGlassBorder(), RoundedCornerShape(19.dp))
                                .clickable { viewModel.toggleDayNight() }
                        ) {
                            val thumbOffset by animateDpAsState(
                                targetValue = if (isNight) 65.dp else 2.dp,
                                animationSpec = spring(stiffness = Spring.StiffnessMedium),
                                label = "thumb"
                            )

                            // Glass sliding thumb
                            Box(
                                modifier = Modifier
                                    .padding(vertical = 2.dp)
                                    .offset(x = thumbOffset)
                                    .width(63.dp)
                                    .fillMaxHeight()
                                    .background(
                                        brush = if (isNight) {
                                            Brush.linearGradient(listOf(Color(0xFF3F51B5), Color(0xFF673AB7)))
                                        } else {
                                            Brush.linearGradient(listOf(Color(0xFFFF9800), Color(0xFFFF5722)))
                                        },
                                        shape = RoundedCornerShape(17.dp)
                                    ),
                                contentAlignment = Alignment.Center
                            ) {
                                Icon(
                                    imageVector = if (isNight) Icons.Default.NightsStay else Icons.Default.WbSunny,
                                    contentDescription = null,
                                    tint = AppTextColor(),
                                    modifier = Modifier.size(14.dp)
                                )
                            }

                            // Underlying Labels
                            Row(
                                modifier = Modifier.fillMaxSize(),
                                horizontalArrangement = Arrangement.SpaceAround,
                                verticalAlignment = Alignment.CenterVertically
                            ) {
                                Text(
                                    text = "Day",
                                    color = if (!isNight) AppTextColor() else AppTextColor().copy(alpha = 0.5f),
                                    fontSize = 11.sp,
                                    fontWeight = FontWeight.Bold
                                )
                                Text(
                                    text = "Night",
                                    color = if (isNight) AppTextColor() else AppTextColor().copy(alpha = 0.5f),
                                    fontSize = 11.sp,
                                    fontWeight = FontWeight.Bold
                                )
                            }
                        }
                    }

                    Spacer(modifier = Modifier.height(16.dp))

                    // MAIN DISHES GRID SECTION
                    Text(
                        text = "Main Dishes (Select One to View Add-ons)",
                        color = AppTextColor(),
                        fontSize = 14.sp,
                        fontWeight = FontWeight.Bold,
                        modifier = Modifier.padding(bottom = 8.dp)
                    )

                    // Standard fluid wrapping flow/grid of frosted tiles
                    Column(verticalArrangement = Arrangement.spacedBy(10.dp)) {
                        val rows = mainDishes.chunked(2)
                        rows.forEach { row ->
                            Row(
                                modifier = Modifier.fillMaxWidth(),
                                horizontalArrangement = Arrangement.spacedBy(10.dp)
                            ) {
                                row.forEach { dish ->
                                    val qty = quantities[dish.id] ?: 0
                                    val isSelected = selectedMainId == dish.id

                                    // Frosted Dish Tile
                                    Box(
                                        modifier = Modifier
                                            .weight(1f)
                                            .height(120.dp)
                                            .bounceClick()
                                            .background(
                                                brush = if (isSelected) {
                                                    Brush.verticalGradient(listOf(Color(0x4000E5FF), Color(0x1F00E5FF)))
                                                } else {
                                                    Brush.verticalGradient(listOf(Color(0x1FFFFFFF), Color(0x09FFFFFF)))
                                                },
                                                shape = RoundedCornerShape(18.dp)
                                            )
                                            .border(
                                                width = if (isSelected) 1.5.dp else 0.5.dp,
                                                color = if (isSelected) AccentCyan else Color(0x2BFFFFFF),
                                                shape = RoundedCornerShape(18.dp)
                                            )
                                            .clip(RoundedCornerShape(18.dp))
                                            .clickable { viewModel.selectMainDish(dish.id) }
                                            .padding(12.dp)
                                    ) {
                                        Column(
                                            modifier = Modifier.fillMaxSize(),
                                            verticalArrangement = Arrangement.SpaceBetween
                                        ) {
                                            Row(
                                                modifier = Modifier.fillMaxWidth(),
                                                horizontalArrangement = Arrangement.SpaceBetween,
                                                verticalAlignment = Alignment.CenterVertically
                                            ) {
                                                Column(modifier = Modifier.weight(1f)) {
                                                    Text(
                                                        text = dish.banglaName,
                                                        color = AppTextColor(),
                                                        fontSize = 14.sp,
                                                        fontWeight = FontWeight.Bold,
                                                        maxLines = 1,
                                                        overflow = TextOverflow.Ellipsis
                                                    )
                                                    Text(
                                                        text = dish.englishName,
                                                        color = AppTextColor().copy(alpha = 0.6f),
                                                        fontSize = 11.sp,
                                                        maxLines = 1,
                                                        overflow = TextOverflow.Ellipsis
                                                    )
                                                }
                                                Text(
                                                    text = "₹${dish.price.toInt()}",
                                                    color = AccentGold,
                                                    fontSize = 14.sp,
                                                    fontWeight = FontWeight.Bold
                                                )
                                            }

                                            // Decrement / Counter UI
                                            Row(
                                                modifier = Modifier.fillMaxWidth(),
                                                horizontalArrangement = Arrangement.End,
                                                verticalAlignment = Alignment.CenterVertically
                                            ) {
                                                if (qty > 0) {
                                                    IconButton(
                                                        onClick = { viewModel.updateItemQuantity(dish.id, -1) },
                                                        modifier = Modifier
                                                            .size(26.dp)
                                                            .background(getGlassIconButtonBg(), CircleShape)
                                                    ) {
                                                        Icon(Icons.Default.Remove, contentDescription = "-", tint = AppTextColor(), modifier = Modifier.size(14.dp))
                                                    }
                                                    Text(
                                                        text = "$qty",
                                                        color = AppTextColor(),
                                                        fontSize = 15.sp,
                                                        fontWeight = FontWeight.Bold,
                                                        modifier = Modifier.padding(horizontal = 8.dp)
                                                    )
                                                }
                                                IconButton(
                                                    onClick = { viewModel.updateItemQuantity(dish.id, 1) },
                                                    modifier = Modifier
                                                        .size(26.dp)
                                                        .background(if (qty > 0) AccentCyan else getGlassIconButtonBg(), CircleShape)
                                                ) {
                                                    Icon(
                                                        imageVector = Icons.Default.Add,
                                                        contentDescription = "+",
                                                        tint = if (qty > 0) Color(0xFF0F0B26) else AppTextColor(),
                                                        modifier = Modifier.size(14.dp)
                                                    )
                                                }
                                            }
                                        }
                                    }
                                }
                                if (row.size < 2) {
                                    Spacer(modifier = Modifier.weight(1f))
                                }
                            }
                        }
                    }

                    Spacer(modifier = Modifier.height(18.dp))

                    // DYNAMICALLY RENDERED ADD-ON DISHES TRAY BELOW
                    AnimatedVisibility(
                        visible = selectedMainId != null,
                        enter = expandVertically(animationSpec = spring(stiffness = Spring.StiffnessMedium)) + fadeIn(),
                        exit = shrinkVertically() + fadeOut()
                    ) {
                        Column {
                            val activeMainName = menuItems.find { it.id == selectedMainId }?.englishName ?: ""
                            Text(
                                text = "Supplementary Add-ons for $activeMainName",
                                color = AccentCyan,
                                fontSize = 14.sp,
                                fontWeight = FontWeight.Bold,
                                modifier = Modifier.padding(bottom = 8.dp)
                            )

                            GlassCard(modifier = Modifier.fillMaxWidth()) {
                                Column(
                                    modifier = Modifier
                                        .fillMaxWidth()
                                        .padding(12.dp),
                                    verticalArrangement = Arrangement.spacedBy(8.dp)
                                ) {
                                    addOnDishes.forEach { addOn ->
                                        val addOnQty = quantities[addOn.id] ?: 0

// CHRONOLOGICAL VIEW MEAL HISTORY MODAL
// ==========================================
@Composable
fun MealHistoryModal(
    viewModel: FinancialViewModel,
    onDismiss: () -> Unit
) {
    val transactions by viewModel.activeHotelTransactions.collectAsState()
    val userProfile by viewModel.userProfile.collectAsState()
    val sdf = SimpleDateFormat("dd MMM, hh:mm a", Locale.getDefault())
    val monthFormat = SimpleDateFormat("MMMM yyyy", Locale.getDefault())
    
    var monthOffset by remember { mutableStateOf(0) }
    
    val filteredTransactions = remember(transactions, monthOffset) {
        val cal = Calendar.getInstance()
        cal.add(Calendar.MONTH, monthOffset)
        val targetMonth = cal.get(Calendar.MONTH)
        val targetYear = cal.get(Calendar.YEAR)
        
        transactions.filter {
            val txCal = Calendar.getInstance()
            txCal.timeInMillis = it.timestamp
            txCal.get(Calendar.MONTH) == targetMonth && txCal.get(Calendar.YEAR) == targetYear
        }
    }
    
    val currentDisplayMonth = remember(monthOffset) {
        val cal = Calendar.getInstance()
        cal.add(Calendar.MONTH, monthOffset)
        monthFormat.format(cal.time)
    }

    Dialog(
        onDismissRequest = onDismiss,
        properties = DialogProperties(usePlatformDefaultWidth = false)
    ) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(getAppBackgroundGradient(userProfile.themeMode))
                .statusBarsPadding()
                .navigationBarsPadding()
        ) {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(16.dp)
            ) {
                // Header
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    IconButton(
                        onClick = onDismiss,
                        modifier = Modifier.background(getGlassBorder(), CircleShape)
                    ) {
                        Icon(Icons.Default.ArrowBack, contentDescription = "Back", tint = AppTextColor())
                    }

                    Text(
                        text = "Meal Record Ledger",
                        color = AppTextColor(),
                        fontSize = 18.sp,
                        fontWeight = FontWeight.Bold
                    )

                    Spacer(modifier = Modifier.width(48.dp)) // Equal space balancing
                }

                Spacer(modifier = Modifier.height(16.dp))

                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    IconButton(onClick = { monthOffset -= 1 }) {
                        Icon(Icons.Default.ArrowBack, "Previous Month", tint = AppTextColor())
                    }
                    Text(currentDisplayMonth, color = AppTextColor(), fontWeight = FontWeight.Bold, fontSize = 16.sp)
                    IconButton(onClick = { monthOffset += 1 }, enabled = monthOffset < 0) {
                        Icon(Icons.Default.ArrowForward, "Next Month", tint = if (monthOffset < 0) AppTextColor() else AppTextColor().copy(alpha = 0.3f))
                    }
                }
                Spacer(modifier = Modifier.height(16.dp))
                
                if (filteredTransactions.isEmpty()) {
                    Box(modifier = Modifier.weight(1f), contentAlignment = Alignment.Center) {
                        EmptyStateCard(message = "No past meal logs for $currentDisplayMonth.")
                    }
                } else {
                    LazyColumn(
                        modifier = Modifier.weight(1f),
                        verticalArrangement = Arrangement.spacedBy(10.dp)
                    ) {
                        items(filteredTransactions) { tx ->
                            GlassCard(modifier = Modifier.fillMaxWidth()) {
                                Row(
                                    modifier = Modifier
                                        .fillMaxWidth()
                                        .padding(14.dp),
                                    horizontalArrangement = Arrangement.SpaceBetween,
                                    verticalAlignment = Alignment.CenterVertically
                                ) {
                                    Column(modifier = Modifier.weight(1f)) {
                                        Row(verticalAlignment = Alignment.CenterVertically) {
                                            Icon(
                                                imageVector = if (tx.isNight) Icons.Default.NightsStay else Icons.Default.WbSunny,
                                                contentDescription = null,
                                                tint = if (tx.isNight) AccentCyan else AccentGold,
                                                modifier = Modifier.size(14.dp)
                                            )
                                            Spacer(modifier = Modifier.width(6.dp))
                                            Text(
                                                text = sdf.format(Date(tx.timestamp)),
                                                color = AppTextColor(),
                                                fontSize = 13.sp,
                                                fontWeight = FontWeight.Bold
                                            )
                                        }
                                        Spacer(modifier = Modifier.height(4.dp))
                                        
                                        // Expandable simple text list of ordered items
                                        val items = TransactionSerializer.deserialize(tx.itemsSerialized)
                                        val summary = items.joinToString(", ") { "${it.name} x${it.quantity}" }
                                        Text(
                                            text = summary,
                                            color = AppTextColor().copy(alpha = 0.6f),
                                            fontSize = 11.sp,
                                            maxLines = 2,
                                            overflow = TextOverflow.Ellipsis
                                        )
                                    }

                                    Row(
                                        verticalAlignment = Alignment.CenterVertically,
                                        horizontalArrangement = Arrangement.End
                                    ) {
                                        Text(
                                            text = "₹${tx.totalAmount.toInt()}",
                                            color = AccentGreen,
                                            fontSize = 15.sp,
                                            fontWeight = FontWeight.Bold,
                                            modifier = Modifier.padding(horizontal = 8.dp)
                                        )

                                        IconButton(
                                            onClick = { viewModel.deleteTransaction(tx) },
                                            modifier = Modifier
                                                .size(28.dp)
                                                .background(Color(0x15FF4081), CircleShape)
                                        ) {
                                            Icon(
                                                imageVector = Icons.Default.Delete,
                                                contentDescription = "Delete",
                                                tint = AccentPink,
                                                modifier = Modifier.size(14.dp)
                                            )
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
    }
}

@Composable
fun EmptyStateCard(message: String) {
    GlassCard(modifier = Modifier.fillMaxWidth()) {
        Column(
            modifier = Modifier
                .fillMaxWidth()
                .padding(24.dp),
            horizontalAlignment = Alignment.CenterHorizontally,
            verticalArrangement = Arrangement.Center
        ) {
            Icon(
                imageVector = Icons.Default.Info,
                contentDescription = null,
                tint = AccentCyan.copy(alpha = 0.6f),
                modifier = Modifier.size(36.dp)
            )
            Spacer(modifier = Modifier.height(10.dp))
            Text(
                text = message,
                color = AppTextColor().copy(alpha = 0.6f),
                fontSize = 13.sp,
                textAlign = TextAlign.Center,
                lineHeight = 18.sp
            )
        }
    }
}

@Composable
fun GlobalSearchModal(viewModel: FinancialViewModel, onDismiss: () -> Unit) {
    var searchQuery by remember { mutableStateOf("") }
    val userProfile by viewModel.userProfile.collectAsState()
    val allExpenses by viewModel.allExtraExpenses.collectAsState()
    val allHotels by viewModel.allHotels.collectAsState()
    val allMenuItems by viewModel.allMenuItems.collectAsState()
    
    val filteredExpenses = remember(searchQuery, allExpenses) {
        if (searchQuery.isBlank()) emptyList()
        else allExpenses.filter { it.name.contains(searchQuery, ignoreCase = true) }
    }
    
    val filteredHotels = remember(searchQuery, allHotels) {
        if (searchQuery.isBlank()) emptyList()
        else allHotels.filter { it.name.contains(searchQuery, ignoreCase = true) }
    }
    
    val filteredMenu = remember(searchQuery, allMenuItems) {
        if (searchQuery.isBlank()) emptyList()
        else allMenuItems.filter { 
            it.englishName.contains(searchQuery, ignoreCase = true) || 
            it.banglaName.contains(searchQuery, ignoreCase = true) 
        }
    }

    Dialog(
        onDismissRequest = onDismiss,
        properties = DialogProperties(usePlatformDefaultWidth = false)
    ) {
        Box(
            modifier = Modifier
                .fillMaxSize()
                .background(getAppBackgroundGradient(userProfile.themeMode))
                .statusBarsPadding()
                .navigationBarsPadding()
        ) {
            Column(
                modifier = Modifier
                    .fillMaxSize()
                    .padding(16.dp)
            ) {
                // Header
                Row(
                    modifier = Modifier.fillMaxWidth(),
                    horizontalArrangement = Arrangement.SpaceBetween,
                    verticalAlignment = Alignment.CenterVertically
                ) {
                    IconButton(
                        onClick = onDismiss,
                        modifier = Modifier.background(getGlassBorder(), CircleShape)
                    ) {
                        Icon(Icons.Default.ArrowBack, contentDescription = "Back", tint = AppTextColor())
                    }
                    Text(
                        text = if (userProfile.language == "bn") "অনুসন্ধান" else "Search",
                        color = AppTextColor(),
                        fontSize = 18.sp,
                        fontWeight = FontWeight.Bold
                    )
                    Spacer(modifier = Modifier.size(48.dp))
                }
                
                Spacer(modifier = Modifier.height(16.dp))
                
                // Search Input
                OutlinedTextField(
                    value = searchQuery,
                    onValueChange = { searchQuery = it },
                    label = { Text(if (userProfile.language == "bn") "এখানে খুঁজুন..." else "Search here...", color = AppTextColor().copy(alpha = 0.6f)) },
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Text),
                    colors = OutlinedTextFieldDefaults.colors(
                        focusedTextColor = AppTextColor(),
                        unfocusedTextColor = AppTextColor(),
                        focusedBorderColor = AccentCyan,
                        unfocusedBorderColor = AppTextColor().copy(alpha = 0.3f)
                    ),
                    modifier = Modifier.fillMaxWidth(),
                    leadingIcon = {
                        Icon(Icons.Default.Search, contentDescription = null, tint = AppTextColor().copy(alpha = 0.5f))
                    },
                    trailingIcon = {
                        if (searchQuery.isNotEmpty()) {
                            IconButton(onClick = { searchQuery = "" }) {
                                Icon(Icons.Default.Close, contentDescription = "Clear", tint = AppTextColor().copy(alpha = 0.5f))
                            }
                        }
                    }
                )
                
                Spacer(modifier = Modifier.height(24.dp))
                
                if (searchQuery.isNotEmpty()) {
                    if (filteredExpenses.isEmpty() && filteredHotels.isEmpty() && filteredMenu.isEmpty()) {
                        Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                            Text(
                                text = if (userProfile.language == "bn") "কোন ফলাফল পাওয়া যায়নি।" else "No results found.",
                                color = AppTextColor().copy(alpha = 0.5f),
                                fontSize = 14.sp
                            )
                        }
                    } else {
                        LazyColumn(
                            modifier = Modifier.fillMaxSize(),
                            verticalArrangement = Arrangement.spacedBy(16.dp)
                        ) {
                            if (filteredHotels.isNotEmpty()) {
                                item {
                                    Text(
                                        text = if (userProfile.language == "bn") "হোটেল" else "Hotels",
                                        color = AppTextColor(),
                                        fontSize = 16.sp,
                                        fontWeight = FontWeight.Bold
                                    )
                                    Spacer(modifier = Modifier.height(8.dp))
                                }
                                items(filteredHotels) { hotel ->
                                    GlassCard(modifier = Modifier.fillMaxWidth()) {
                                        Text(text = hotel.name, color = AppTextColor(), fontWeight = FontWeight.Medium, modifier = Modifier.padding(16.dp))
                                    }
                                }
                            }
                            
                            if (filteredMenu.isNotEmpty()) {
                                item {
                                    Text(
                                        text = if (userProfile.language == "bn") "খাবার" else "Menu Items",
                                        color = AppTextColor(),
                                        fontSize = 16.sp,
                                        fontWeight = FontWeight.Bold
                                    )
                                    Spacer(modifier = Modifier.height(8.dp))
                                }
                                items(filteredMenu) { menu ->
                                    GlassCard(modifier = Modifier.fillMaxWidth()) {
                                        Row(
                                            modifier = Modifier.fillMaxWidth().padding(16.dp),
                                            horizontalArrangement = Arrangement.SpaceBetween
                                        ) {
                                            Text(text = if (userProfile.language == "bn") menu.banglaName else menu.englishName, color = AppTextColor(), fontWeight = FontWeight.Medium)
                                            Text(text = "₹${menu.price}", color = AppTextColor(), fontWeight = FontWeight.Bold)
                                        }
                                    }
                                }
                            }
                            
                            if (filteredExpenses.isNotEmpty()) {
                                item {
                                    Text(
                                        text = if (userProfile.language == "bn") "খরচ" else "Expenses",
                                        color = AppTextColor(),
                                        fontSize = 16.sp,
                                        fontWeight = FontWeight.Bold
                                    )
                                    Spacer(modifier = Modifier.height(8.dp))
                                }
                                items(filteredExpenses) { expense ->
                                    GlassCard(modifier = Modifier.fillMaxWidth()) {
                                        Row(
                                            modifier = Modifier.fillMaxWidth().padding(14.dp),
                                            horizontalArrangement = Arrangement.SpaceBetween,
                                            verticalAlignment = Alignment.CenterVertically
                                        ) {
                                            Column {
                                                Text(
                                                    text = expense.name,
                                                    color = AppTextColor(),
                                                    fontSize = 14.sp,
                                                    fontWeight = FontWeight.Bold
                                                )
                                                Spacer(modifier = Modifier.height(2.dp))
                                                val sdf = java.text.SimpleDateFormat("dd MMM, hh:mm a", java.util.Locale.getDefault())
                                                Text(
                                                    text = sdf.format(java.util.Date(expense.timestamp)),
                                                    color = AppTextColor().copy(alpha = 0.5f),
                                                    fontSize = 11.sp
                                                )
                                            }
                                            Text(
                                                text = "-₹${expense.amount}",
                                                color = Color(0xFFFF5252),
                                                fontWeight = FontWeight.Bold,
                                                fontSize = 16.sp
                                            )
                                        }
                                    }
                                }
                            }
                        }
                    }
                } else {
                    Box(modifier = Modifier.fillMaxSize(), contentAlignment = Alignment.Center) {
                        Text(
                            text = if (userProfile.language == "bn") "হোটেল, খাবার বা খরচ খুঁজতে শুরু করুন।" else "Start typing to search for hotels, meals, or expenses.",
                            color = AppTextColor().copy(alpha = 0.4f),
                            fontSize = 14.sp,
                            textAlign = TextAlign.Center
                        )
                    }
                }
            }
        }
    }
}
